<template>
  <div class="calendar-container">
    <div class="calendar-label">Calendar</div>
    <div class="calendar-header">
      <button @click="prevMonth">‹</button>
      <span>{{ monthName }} {{ currentYear }}</span>
      <button @click="nextMonth">›</button>
    </div>

    <div class="calendar-grid">
      <div class="day-header" v-for="day in daysOfWeek" :key="day">
        {{ day }}
      </div>

      <div
        class="day-cell"
        v-for="(day, index) in calendarDays"
        :key="index"
        :class="{ 'not-current-month': !day.currentMonth }"
      >
        <div class="day-number">{{ day.date.getDate() }}</div>

        <div
          v-for="event in eventsForDay(day.date)"
          :key="event.id"
          class="event-bar"
          :style="eventStyle(event)"
          @click="openEvent(event)"
        >
          <span class="event-time">({{ event.time }}) </span>
          <span class="event-title">{{ event.title }}</span>
        </div>
      </div>
    </div>

    <!-- New Modal Component -->
    <EventModal
      v-if="selectedEvent"
      :event="selectedEvent"
      @close="closeModal"
      @download="downloadCalendar"
    />
  </div>
</template>

<script>
import EventModal from "@/components/MeetingBoard/EventModal.vue";

export default {
  name: "AdvancedCalendar",
  components: {
    EventModal,
  },
  props: {
    events: {
      type: Array,
      default: () => [],
    },
  },
  data() {
    return {
      currentMonth: new Date().getMonth(),
      currentYear: new Date().getFullYear(),
      selectedEvent: null,
      daysOfWeek: [
        "Sunday",
        "Monday",
        "Tuesday",
        "Wednesday",
        "Thursday",
        "Friday",
        "Saturday",
      ],
    };
  },
  computed: {
    monthName() {
      return new Date(this.currentYear, this.currentMonth).toLocaleString(
        "default",
        { month: "long" }
      );
    },
    calendarDays() {
      const start = new Date(this.currentYear, this.currentMonth, 1);
      const end = new Date(this.currentYear, this.currentMonth + 1, 0);
      const days = [];

      const startDay = start.getDay();
      const totalDays = end.getDate();

      // Leading days from previous month
      for (let i = 0; i < startDay; i++) {
        const date = new Date(
          this.currentYear,
          this.currentMonth,
          -(startDay - 1 - i)
        );
        days.push({ date, currentMonth: false });
      }

      // Current month days
      for (let d = 1; d <= totalDays; d++) {
        days.push({
          date: new Date(this.currentYear, this.currentMonth, d),
          currentMonth: true,
        });
      }

      // Trailing days from next month
      const totalCells = Math.ceil(days.length / 7) * 7;
      for (let i = days.length; i < totalCells; i++) {
        const dayOffset = i - days.length;
        const date = new Date(
          this.currentYear,
          this.currentMonth + 1,
          1 + dayOffset
        );
        days.push({ date, currentMonth: false });
      }

      return days;
    },
  },
  methods: {
    nextMonth() {
      if (this.currentMonth === 11) {
        this.currentMonth = 0;
        this.currentYear++;
      } else {
        this.currentMonth++;
      }
    },
    prevMonth() {
      if (this.currentMonth === 0) {
        this.currentMonth = 11;
        this.currentYear--;
      } else {
        this.currentMonth--;
      }
    },
    eventsForDay(date) {
      return this.events.filter(
        (event) => date >= event.start && date <= event.end
      );
    },
    eventStyle(event) {
      return {
        background: event.color,
        color: "#fff",
        padding: "2px 4px",
        borderRadius: "4px",
        fontSize: "12px",
        marginBottom: "2px",
        cursor: "pointer",
      };
    },
    openEvent(event) {
      this.selectedEvent = event;
    },
    closeModal() {
      this.selectedEvent = null;
    },
    downloadCalendar() {
      // implement download logic here
    },
  },
};
</script>
